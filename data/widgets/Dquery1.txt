<script>
/*转载自 https://wiki.biligame.com/evertale/MediaWiki:Dquery.js*/
let queryProperty = paramsFormmatter($("#queryProperty").text(), "|?");
let defaultSearch = $("#defaultSearch").text();
let queryTemplate = $("#queryTemplate").text();
let headerText = $("#headerText").text();
let footerText = $("#footerText").text();
function getconditions() {
	//遍历data-type的元素，生成对应的分类查询参数
	let CategoryCondition = $("[data-type='Category']").find(".active, .DQselectd, input")
		.map(function () {
			let value;
			if ($(this).hasClass('active')) {
				value = $(this).data('conditions');
			} else if ($(this).hasClass('DQselectd') && $(this).val() === "全部") {
				value = null;
			} else {
				value = $(this).val() || $(this).text();
			}
			return value ? `[[分类:${value}]]` : null;
		}).get().filter(Boolean).join("");
	//遍历data-type的元素，生成对应的属性查询参数
	let PropertyCondition = $("[data-type='Property']").find(".active, .DQselectd, input")
		.map(function () {
			let key = $(this).attr("data-property") || $(this).parent().attr("data-property");
			let value = $(this).hasClass('active') ? $(this).data('conditions') : ($(this).val() || $(this).text());
			if (value && !($(this).hasClass('DQselectd') && value === "全部")) {
				if ($(this).is('input')) {
					return { key, value: "~*" + value + "*" };
				} else {
					return { key, value };
				}
			}
		})
		.get()
		.reduce((acc, { key, value }) => {
			if (!acc[key]) acc[key] = [];
			acc[key].push(value);
			return acc;
		}, {});
	let PropertyConditionlength = Object.keys(PropertyCondition).length;
	PropertyCondition = Object.entries(PropertyCondition)
		.map(([key, value]) => `[[${key}::${value.join('||')}]]`)
		.join('');
	if (CategoryCondition === "" && PropertyCondition === "") {
		alert("没有指定查询条件");
	}
	else if (PropertyConditionlength > 7) {
		alert("超出查询条件，最多支持7个属性查询");
	}
	else {
		Condition = CategoryCondition + PropertyCondition;
	}
	return Condition;
}
function queryData(conditions, limit) {
	$.getJSON(mw.util.wikiScript('api'), {
		format: 'json',
		action: 'parse',
		text: '{' + '{#ask:' + conditions + '|format=count}}',
		contentmodel: 'wikitext'
	}).done(function (data) {
		count = parseInt($(data.parse.text["*"]).text());
		$("#queryCount").text(count);
		$("#queryLimit").text(limit);
		refreshPager(count, limit, 1);
		getPageData(conditions, limit, 1);
	});
}
function getPageData(conditions, limit, page) {
	limit = Number(limit);
	page = Number(page);
	let sort = $("[data-type='sort']").find(".active, .DQselectd").text() || "";
	let orderText = $("[data-type='order']").find(".active, .DQselectd").text();
	let order = orderText.replaceAll("降序", "desc").replaceAll("升序", "asc").replaceAll("随机", "rand");
	let parameters = "|template=" + queryTemplate + "|sort=" + sort + "|order=" + order + "|limit=" + limit + "|offset=" + (page - 1) * limit + "|headers=hide|searchlabel=|format=template|link=none";
	let queryText = headerText + '{{#ask:' + conditions + queryProperty + parameters + '}}' + footerText;
	$.getJSON(mw.util.wikiScript('api'), {
		format: 'json',
		action: 'parse',
		text: queryText,
		contentmodel: 'wikitext'
	}).done(function (data) {
		$("#queryDataGrid").html(data.parse.text["*"]);
     // ✅ 只为Dquery结果中的元素设置勾选状态
     // 【新增】结果内所有链接统一新标签页打开（带安全 rel）
		$("#queryDataGrid a").attr({
			target: "_blank",
			rel: "noopener noreferrer"
		});
    initializeDqueryCollections();
	});
}

// ✅ 新增函数：选择性初始化Dquery中的勾勾册
function initializeDqueryCollections() {
  const $items = $("#queryDataGrid .bwiki-collection");
  if ($items.length === 0) return;

  // 收集所有数据集名称
  const listNames = [...new Set($items.map((_, el) => $(el).attr("data-collectionlist")).get())];
  if (listNames.length === 0) return;

  const api = new mw.Api();
  
  // 获取用户选项
  api.get({
    "action": "query",
    "format": "json",
    "meta": "userinfo",
    "formatversion": "2",
    "uiprop": "options"
  }).then(function(data) {
    const options = data.query.userinfo.options;
    
    // 为每个元素设置状态
    $items.each(function() {
      const $item = $(this);
      const listName = $item.attr("data-collectionlist");
      const name = $item.attr("data-collection");
      
      if (!listName || !name) return;
      
      // 检查是否已勾选
      const checkedItems = (options["userjs-collectionList-" + listName] || "").split(",").filter(Boolean);
      const isChecked = checkedItems.includes(name);
      
      $item.toggleClass("checked", isChecked)
        .attr("data-paramc", isChecked ? "已获取" : "未获取");
      
      // 确保勾选框存在
      const $checkbox = $item.find(".bwiki-collection-checkbox");
      if ($checkbox.length === 0) {
        $item.append('<div class="bwiki-collection-checkbox checkbox-default"><div></div></div>');
      } else if ($checkbox.find("div").length === 0) {
        $checkbox.append("<div>");
      }
    });
    
    // ✅ 绑定点击事件（命名空间'dquery'避免冲突）
    $(document).off("click.dquery", "#queryDataGrid .bwiki-collection-checkbox div")
      .on("click.dquery", "#queryDataGrid .bwiki-collection-checkbox div", function() {
        const $item = $(this).closest(".bwiki-collection");
        const listName = $item.attr("data-collectionlist");
        const name = $item.attr("data-collection");
        
        if (!listName || !name) return;
        
        // 读取-修改-保存勾选状态
        const optionKey = "userjs-collectionList-" + listName;
        
        api.get({
          "action": "query",
          "format": "json",
          "meta": "userinfo",
          "formatversion": "2",
          "uiprop": "options"
        }).then(function(data) {
          const currentValue = data.query.userinfo.options[optionKey] || "";
          let list = currentValue.split(",").filter(Boolean);
          const index = list.indexOf(name);
          
          if (index === -1) {
            // 勾选
            list.push(name);
            $item.addClass("checked").attr("data-paramc", "已获取");
          } else {
            // 取消勾选
            list.splice(index, 1);
            $item.removeClass("checked").attr("data-paramc", "未获取");
          }
          
          // 保存到用户选项
          api.postWithToken('csrf', {
            "action": "options",
            "format": "json",
            "optionname": optionKey,
            "optionvalue": list.join(",")
          });
          
          // 更新计数（如果有计数器的话）
          $(".bwiki-collection-checknum-" + listName).text(list.length);
        });
      });
  });
}

function refreshPager(count, limit, page) {
	var sumPage = Math.min(Math.ceil(Number(count) / Number(limit)));
	page = Number(page);
	page = page === 0 ? 1 : page;
	page = page < 0 ? sumPage : page;
	var pageStart = Math.max(1, page - 2);
	var pageEnd = Math.min(Math.ceil(count / limit), pageStart + 4);
	pageStart = pageEnd <= 5 ? 1 : pageStart;
	var $a = $("<a>").attr({ href: "javascript:void(0);" });
	var btnArray = [];
	for (var i = pageStart; i <= pageEnd; i++) {
		btnArray.push($("<li>").attr({ class: "gridPageBtn" + (page === i ? " active" : "") }).append($a.clone().text(i)));
	}
	$(".gridPageBtn").remove();
	$(".pagination li:first-child").after(btnArray);
	$(".gridFirstPageBtn")[page === 1 ? "addClass" : "removeClass"]("hide");
	$(".gridLastPageBtn")[page === sumPage || pageEnd === 1 ? "addClass" : "removeClass"]("hide");
	document.body.scrollTop = $("#queryDataGrid").offset().top - 100;
	return page;
}
function paramsFormmatter(param, connector) {
	return ("、" + param).split("、").join(connector);
}
(window.RLQ = window.RLQ || []).push([["jquery"], function () {
	$(function () {
		//隐藏#JSerror元素
		$("#JSerror").addClass("hide");
		//显示#queryData元素，移出#queryData元素的.hide类
		$("#queryData").removeClass("hide");
		//.DQinput元素插入input，内容为.DQinput元素中的内容
		$(".DQinput").each(function () {
			$(this).replaceWith('<input value="' + $(this).text() + '" class="DQinput form-control" data-conditions="' + $(this).text() + '" style="width:auto;" placeholder="支持模糊查询"/>');
		});
		//.DQinput元素的值改变时，同步改变属性data-conditions的值
		$(".DQinput").on('input', function () {
			$(this).attr("data-conditions", $(this).val());
		});
		//.DQselect下级子元素中包含的li元素，将点击的li元素中的内容赋值给.DQselect的.DQselectd
		$(".DQselect li").click(function () {
			$(this).parent().parent().find(".DQselectd").text($(this).text());
			$(this).parent().parent().find(".DQselectd").attr("data-conditions", $(this).text());

		});
		//点击.Single中的li元素时，移除其他li元素上的active类
		$(".Single div").click(function () {
			$(this).siblings().removeClass("active");
		});
		$("#queryData").click(function () {
			let Condition = getconditions();
			let limit = $("[data-type='limit']").find(".DQselectd").text() || 200;
			queryData(Condition, limit);
		});
		//点击#clearConditions按钮，清空所有除了.disabled按钮的.active类，删除所有DQinput的值，将所有.DQselect的值设置为"全部"
		$("#clearConditions").click(function () {
			// 清空所有除了.disabled和.hide子元素按钮的.active类
			$(".active").not(".disabled, .hide *").removeClass("active");
			// 删除所有DQinput的值
			$(".DQinput").val("").attr("data-conditions", "");
			// 将所有data-type为Property或Category的.DQselect值设置为"全部"
			$("[data-type='Property'], [data-type='Category']").find(".DQselectd").text("全部").attr("data-conditions", "全部");
		});
		// 绑定点击事件到body，用于处理带有“queryParams”类的按钮点击
		$("body")
			// 绑定点击事件到清除参数按钮，点击时移除所有选中项的“selected”类
			.on("click", "#clearParamBtn", function () {
				$(".selected").removeClass("selected")
			})
			// 绑定点击事件到分页按钮，点击时根据按钮数据获取页面信息，并更新显示内容
			.on("click", ".gridPageBtn, .gridFirstPageBtn, .gridLastPageBtn", function () {
				// 如果按钮处于隐藏或激活状态，则不进行任何操作
				if ($(this).hasClass("hide") || $(this).hasClass("active")) return false;
				// 获取按钮的数据页码或文本页码
				var page = $(this).attr("data-page") || $(this).text();
				// 使用获取的页码更新页面数据
				let limit = $("[data-type='limit']").find(".DQselectd").text() || 200;
				let Condition = getconditions();
				getPageData(Condition, limit, refreshPager(count, limit, page));
			});
		// 根据默认搜索配置，自动执行查询数据操作
		$(document).ready(function () {
			let Condition = getconditions();
			let limit = $("[data-type='limit']").find(".DQselectd").text() || 200;
			if (defaultSearch === "true" || defaultSearch === "1" || defaultSearch === "查询") queryData(Condition, limit);
		});
	})
}])
